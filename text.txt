ALTER TABLE `receipts`
ADD COLUMN `is_receive` TINYINT(1) NOT NULL DEFAULT 0 AFTER `is_invoice`,
ADD COLUMN `is_bill` TINYINT(1) NOT NULL DEFAULT 0 AFTER `is_receive`,
ADD COLUMN `po_no` VARCHAR(255) NULL UNIQUE AFTER `is_bill`,
ADD COLUMN `grn_no` VARCHAR(255) NULL UNIQUE AFTER `po_no`,
ADD COLUMN `purchase_bill_no` VARCHAR(255) NULL UNIQUE AFTER `grn_no`;

CREATE TABLE `products` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `company_id` BIGINT UNSIGNED NULL,
    `category_id` BIGINT UNSIGNED NULL,
    `product_code` VARCHAR(255) NOT NULL UNIQUE,
    `name` VARCHAR(255) NOT NULL,
    `barcode` VARCHAR(255) NULL,
    `sku` VARCHAR(255) NULL,
    `unit` VARCHAR(255) NOT NULL DEFAULT 'PCS',
    `purchase_price` DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    `sale_price` DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    `current_stock` DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    `minimum_stock` DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    `description` TEXT NULL,
    `status` ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active',
    `created_by` BIGINT UNSIGNED NULL,
    `created_at` TIMESTAMP NULL DEFAULT NULL,
    `updated_at` TIMESTAMP NULL DEFAULT NULL,
    PRIMARY KEY (`id`),
    CONSTRAINT `products_company_id_foreign`
        FOREIGN KEY (`company_id`) REFERENCES `companies`(`id`)
        ON DELETE CASCADE,
    CONSTRAINT `products_category_id_foreign`
        FOREIGN KEY (`category_id`) REFERENCES `categories`(`id`)
        ON DELETE SET NULL,
    CONSTRAINT `products_created_by_foreign`
        FOREIGN KEY (`created_by`) REFERENCES `users`(`id`)
        ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `stock_transactions` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    `company_id` BIGINT UNSIGNED NULL,
    `branch_id` BIGINT UNSIGNED NULL,
    `product_id` BIGINT UNSIGNED NOT NULL,
    `receipt_item_id` BIGINT UNSIGNED NULL,
    `transaction_type` ENUM('Purchase', 'Sale', 'Adjustment', 'Opening') NOT NULL,
    `stock_in` DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    `stock_out` DECIMAL(15,2) NOT NULL DEFAULT 0.00,
    `balance` DECIMAL(15,2) NOT NULL,
    `transaction_date` DATE NOT NULL,
    `created_by` BIGINT UNSIGNED NULL,
    `created_at` TIMESTAMP NULL DEFAULT NULL,
    `updated_at` TIMESTAMP NULL DEFAULT NULL,

    PRIMARY KEY (`id`),

    CONSTRAINT `stock_transactions_company_id_foreign`
        FOREIGN KEY (`company_id`)
        REFERENCES `companies`(`id`)
        ON DELETE CASCADE,

    CONSTRAINT `stock_transactions_branch_id_foreign`
        FOREIGN KEY (`branch_id`)
        REFERENCES `branches`(`id`)
        ON DELETE SET NULL,

    CONSTRAINT `stock_transactions_product_id_foreign`
        FOREIGN KEY (`product_id`)
        REFERENCES `products`(`id`)
        ON DELETE CASCADE,

    CONSTRAINT `stock_transactions_receipt_item_id_foreign`
        FOREIGN KEY (`receipt_item_id`)
        REFERENCES `receipt_items`(`id`)
        ON DELETE SET NULL,

    CONSTRAINT `stock_transactions_created_by_foreign`
        FOREIGN KEY (`created_by`)
        REFERENCES `users`(`id`)
        ON DELETE SET NULL
) ENGINE=InnoDB
DEFAULT CHARSET=utf8mb4
COLLATE=utf8mb4_unicode_ci;
